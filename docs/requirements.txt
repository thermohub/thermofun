#   PYTHONPATH=<pypkg> LD_PRELOAD=<build>/ThermoFun/libThermoFun.so python -m sphinx -W --keep-going -b html docs <site>
#   python3 -m http.server 8000 --bind 127.0.0.1 --directory <site>   # then open http://127.0.0.1:8000/
